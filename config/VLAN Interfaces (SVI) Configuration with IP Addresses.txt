# ============================================
# CONFIGURE VLAN INTERFACES (SVIs)
# ============================================

# Set VLAN 10 IP Gateway (Sales & Marketing)
curl -u $ODL_AUTH -X POST \
  -H "Content-Type: application/json" \
  -d '{
    "input": {
      "tenant-name": "NAWI",
      "bridge-name": "VLAN10_Sales_Marketing",
      "cidr": "10.10.10.1/24"
    }
  }' \
  $ODL_URL/restconf/operations/vtn:add-vbridge-interface
